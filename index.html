Операторы (арифметика, логика, побитовые)
Все операторы языка Python, включая целочисленные и сравнения идентичности.
🐍 python 
# Арифметика
10 / 3     # 3.3333... (всегда float)
10 // 3    # 3 (целочисленное деление, округление вниз к -inf)
-10 // 3   # -4 ! (важно: Python округляет к минус бесконечности)
10 % 3     # 1 (остаток от деления)
2 ** 8     # 256 (возведение в степень)

# Сравнение значений и идентичности
a == b     # равны ли значения объектов
a != b     # не равны
a is b     # указывают ли переменные на один и тот же объект в памяти (id(a) == id(b))
a is None  # ПРАВИЛЬНО: проверку на None всегда делать через is, а не ==

# Побитовые операции
5 & 3      # 1 (побитовое И / AND)
5 | 3      # 7 (побитовое ИЛИ / OR)
5 ^ 3      # 6 (побитовое XOR)
~5         # -6 (побитовое НЕ / NOT: -(x + 1))
1 << 4     # 16 (побитовый сдвиг влево: умножение на 2^4)
16 >> 2    # 4 (побитовый сдвиг вправо: деление на 2^2)
Моржовый оператор := (Walrus Operator) PYTHON 3.8+
Присваивание внутри выражения (Python 3.8+). Позволяет вычислить значение, сохранить в переменную и сразу использовать.
🐍 python 
# Пример 1: чтение потока или строк до условия
while (line := input_stream.readline()):
    process(line)

# Пример 2: проверка длины без повторного вычисления
data = fetch_results()
if (n := len(data)) > 0:
    print(f"Найдено {n} элементов")

# Пример 3: в list comprehension (избегаем двойного вызова тяжелой функции)
results = [y for x in items if (y := heavy_calc(x)) > 100]
Моржовый оператор := сокращает дублирование кода и повышает производительность, исключая повторные вычисления.
Изменяемые vs Неизменяемые (Mutable vs Immutable)
Критическая концепция: изменяемые объекты меняются по ссылке. Неизменяемые создают новый объект.
🐍 python 
a = [1, 2, 3]
b = a            # b ссылается на ТОТ ЖЕ список!
b.append(4)
print(a)         # [1, 2, 3, 4] — список 'a' тоже изменился!

# Как сделать независимую копию:
c = a.copy()     # или a[:] — поверхностная копия (shallow copy)

# Глубокая копия для вложенных структур:
import copy
deep = copy.deepcopy(nested_structure)
Никогда не используйте изменяемые объекты (списки [], словари {}) как значения по умолчанию в аргументах функций! Используйте arg=None и создавайте объект внутри.

Строки и Текст

Основы языка
Форматирование f-strings (PRO фишки) ТОП
Современные f-строки поддерживают отладку, форматирование чисел, дат и выравнивание.
🐍 python 
name = "Python"
ver = 3.12
pi = 3.14159265
price = 1250000.75

# 1. Отладка переменной (Python 3.8+)
print(f"{name=}, {ver=}")     # name='Python', ver=3.12

# 2. Округление и точность float
print(f"{pi:.2f}")            # '3.14'
print(f"{price:,.2f}")        # '1,250,000.75' (разделитель тысяч)
print(f"{price:_}")           # '1_250_000.75' (разделитель _)

# 3. Проценты и системы счисления
print(f"{0.756:.1%}")         # '75.6%'
print(f"{255:#x}, {255:#b}")  # '0xff', '0b11111111'

# 4. Выравнивание и паддинг
print(f"{'лево':<10}")        # 'лево      '
print(f"{'право':>10}")       # '     право'
print(f"{'центр':^10}")       # '  центр   '
print(f"{42:05d}")            # '00042' (дополнение нулями)

# 5. Дата и время напрямую в f-строке
from datetime import datetime
now = datetime.now()
print(f"{now:%Y-%m-%d %H:%M}") # '2026-11-11 18:45'
Срезы строк (Slicing)
Синтаксис: s[start:stop:step]. Правая граница всегда НЕ включительно.
🐍 python 
s = "Hello, World!"
s[0]         # 'H' — первый символ
s[-1]        # '!' — последний символ
s[0:5]       # 'Hello' (с 0 по 4 индекс)
s[7:]        # 'World!' (с 7 до самого конца)
s[:5]        # 'Hello' (с начала до 5)
s[::2]       # 'Hlo ol!' (каждый 2-й символ)
s[::-1]      # '!dlroW ,olleH' (реверс строки в один миг!)
s[1:-1]      # 'ello, World' (срез без первого и последнего символа)
Методы строк (Полная таблица)
Строки в Python неизменяемы (immutable). Любой метод возвращает НОВУЮ строку.
МетодОписаниеПример
.split(sep, maxsplit)Разбить строку в список"a b c".split() → ['a','b','c']
.join(iterable)Склеить список строк через разделитель"-".join(['a','b']) → "a-b"
.replace(old, new, count)Замена подстроки"1-2-3".replace("-",":",1) → "1:2-3"
.strip([chars])Очистить символы по краям" xx ".strip() → "xx"
.partition(sep)Разбивает на (до, sep, после)"user@host".partition("@") → ('user','@','host')
.removeprefix(p)Удалить префикс (3.9+)"v1.2".removeprefix("v") → "1.2"
🐍 python 
# Удаление префиксов и суффиксов (Python 3.9+)
"https://python.org".removeprefix("https://") # 'python.org'
"report.pdf".removesuffix(".pdf")             # 'report'

# Разделение и объединение
parts = "apple,banana,cherry".split(",")     # ['apple', 'banana', 'cherry']
joined = " | ".join(parts)                   # 'apple | banana | cherry'

# Поиск и замена
"hello world".replace("world", "python")     # 'hello python'
"hello".startswith("he")                     # True
"hello".endswith("lo")                       # True
"hello".find("ll")                           # 2 (индекс первого совпадения, -1 если нет)

# Очистка и регистр
"  тест 
".strip()                          # 'тест' (удаляет пробелы и переводы строк)
"python".upper()                             # 'PYTHON'
"PYTHON".lower()                             # 'python'
"ß".casefold()                               # 'ss' (мощнее lower() для интернационализации)
"hello world".title()                        # 'Hello World'
"123".isdigit()                              # True
"abc".isalpha()                              # True

Структуры данных
Методы списка (list)
Список — динамический изменяемый массив ссылок на объекты.
ОперацияСложность по времениОписание
lst[i] / lst[i] = xO(1)Доступ / замена по индексу
lst.append(x)O(1) амортизированноДобавление в конец
lst.pop()O(1)Удаление с конца
lst.pop(0) / lst.insert(0, x)O(n)Удаление/вставка в начало (медленно! Используйте collections.deque)
x in lstO(n)Линейный поиск элемента
lst.sort()O(n log n)Алгоритм Timsort (стабильная сортировка)
🐍 python 
nums = [10, 20, 30]

# Добавление элементов
nums.append(40)             # [10, 20, 30, 40] — O(1) в конец
nums.insert(1, 15)          # [10, 15, 20, 30, 40] — O(n) по индексу
nums.extend([50, 60])       # расширить списком (или nums += [50, 60])

# Удаление элементов
val = nums.pop()            # удаляет и возвращает последний (60) — O(1)
val = nums.pop(0)           # удаляет первый элемент (10) — O(n)
nums.remove(20)             # удаляет первое вхождение значения 20 (ValueError если нет)
nums.clear()                # очищает весь список []
del nums[1:3]               # удаляет диапазон элементов

# Поиск и счет
idx = ["a", "b", "c"].index("b")  # 1
cnt = [1, 2, 1, 1].count(1)        # 3
Сортировка: .sort() vs sorted()
.sort() меняет список на месте. sorted() возвращает новый список.
🐍 python 
users = [
    {"name": "Алексей", "age": 25},
    {"name": "Мария", "age": 19},
    {"name": "Дмитрий", "age": 31}
]

# Сортировка по ключу (лямбда)
sorted_by_age = sorted(users, key=lambda u: u["age"])

# Сортировка по убыванию через reverse=True
sorted_desc = sorted(users, key=lambda u: u["age"], reverse=True)

# Использование operator.itemgetter (быстрее лямбды)
from operator import itemgetter
users.sort(key=itemgetter("age"))
Распаковка списков и кортежей (*rest)
Оператор звездочки для захвата остатка элементов.
🐍 python 
first, *middle, last = [1, 2, 3, 4, 5]
# first = 1
# middle = [2, 3, 4]
# last = 5

head, *tail = [10, 20, 30]
# head = 10, tail = [20, 30]

# Слияние списков через распаковку
a = [1, 2]
b = [3, 4]
merged = [*a, 99, *b]  # [1, 2, 99, 3, 4]

Кортежи (tuple)

Структуры данных
Кортежи (tuple) ВАЖНО
Неизменяемая (immutable) упорядоченная последовательность. Занимает меньше памяти, чем list, и может быть ключом в словаре.
🐍 python 
# Создание кортежа
point = (10, 20)
single = (42,)          # Запятая обязательна! Без нее (42) — это просто число 42!

# Своп переменных без временной переменной
a, b = 1, 2
a, b = b, a            # a=2, b=1 в одну строку!

# Распаковка кортежа
x, y = point
Именованные кортежи (namedtuple & typing.NamedTuple)
Позволяют обращаться к полям по имени (как к атрибутам класса), сохраняя легковесность кортежа.
🐍 python 
from collections import namedtuple

Point = namedtuple("Point", ["x", "y"])
p = Point(10, 20)
print(p.x, p.y)        # 10 20 (обращение по имени!)
print(p[0], p[1])      # 10 20 (и по индексу тоже работает)

# Современный вариант с типами (typing.NamedTuple)
from typing import NamedTuple

class User(NamedTuple):
    id: int
    username: str
    is_active: bool = True

u = User(1, "emil")
print(u.username)      # 'emil'

Множества (set)

Структуры данных
Множества (set) и frozenset
Коллекция уникальных хэшируемых элементов. Проверка вхождения x in s выполняется за O(1).
🐍 python 
s = {1, 2, 3, 3, 3}    # {1, 2, 3} — дубликаты отсекаются автоматически
empty = set()             # ПУСТОЕ множество! {} создает пустой dict!

s.add(4)                 # добавить элемент
s.discard(10)            # удалить без ошибки если нет (в отличие от remove)

# frozenset — неизменяемый аналог
fs = frozenset([1, 2, 3])
dict_with_set_key = {fs: "значение"} # обычный set ключом быть НЕ может!
Математические операции над множествами
Быстрые теоретико-множественные операции через операторы.
🐍 python 
a = {1, 2, 3}
b = {2, 3, 4}

a | b      # {1, 2, 3, 4} — Объединение (Union)
a & b      # {2, 3} — Пересечение (Intersection)
a - b      # {1} — Разность (Difference, элементы только в a)
a ^ b      # {1, 4} — Симметричная разность (XOR, без общих)

# Проверки подмножеств
{1, 2}.issubset(a)       # True (или {1, 2} <= a)
a.issuperset({1, 2})     # True (или a >= {1, 2})
a.isdisjoint({99, 100})  # True (нет общих элементов)

Словари (dict)

Структуры данных
Словари (dict): создание и безопасная работа MUST KNOW
Хэш-таблица. С Python 3.7 гарантированно сохраняет порядок вставки ключей.
🐍 python 
d = {"name": "Мария", "age": 22}

# Безопасное чтение с дефолтом (не выбрасывает KeyError!)
role = d.get("role", "user")       # 'user'

# setdefault: возвращает значение, а если нет ключа — добавляет с дефолтом
d.setdefault("tags", []).append("admin")

# Слияние словарей в Python 3.9+ через оператор |
d1 = {"a": 1, "b": 2}
d2 = {"b": 99, "c": 3}
merged = d1 | d2                   # {'a': 1, 'b': 99, 'c': 3}
d1 |= d2                           # обновление на месте (in-place)

# Удаление элементов
val = d.pop("age", None)           # удаляет и возвращает значение
k, v = d.popitem()                 # удаляет и возвращает последний (key, value)
Специальные словари (collections)
Counter, defaultdict — экономят десятки строк вспомогательного кода.
🐍 python 
from collections import defaultdict, Counter

# 1. defaultdict — автоматически инициализирует ключ при первом обращении
group = defaultdict(list)
group["users"].append("Alice")    # не нужно проверять if "users" in group!

counts = defaultdict(int)
counts["views"] += 1              # по умолчанию 0

# 2. Counter — быстрый подсчет частоты элементов
words = ["яблоко", "банан", "яблоко", "груша", "банан", "яблоко"]
c = Counter(words)
print(c)                          # Counter({'яблоко': 3, 'банан': 2, 'груша': 1})
print(c.most_common(2))           # [('яблоко', 3), ('банан', 2)] — топ-2

Компрехеншены

Структуры данных
Списковые, словарные и множественные включения
Компактный и идиоматичный способ создания коллекций. Выполняются на уровне языка C быстрее циклов for.
🐍 python 
# 1. List comprehension: [выражение for x in iter if условие]
squares = [x**2 for x in range(10) if x % 2 == 0]
# [0, 4, 16, 36, 64]

# if-else перед for: [A if cond else B for x in iter]
parity = ["четное" if x % 2 == 0 else "нечетное" for x in range(4)]
# ['четное', 'нечетное', 'четное', 'нечетное']

# 2. Dict comprehension: {k: v for x in iter}
names = ["alex", "maria", "john"]
len_map = {name: len(name) for name in names}
# {'alex': 4, 'maria': 5, 'john': 4}

# Инвертирование словаря (значения становятся ключами)
inverted = {v: k for k, v in len_map.items()}

# 3. Set comprehension: {выражение for x in iter}
unique_lens = {len(w) for w in names}  # {4, 5}

# 4. Вложенный comprehension (разворачивание 2D матрицы в 1D)
matrix = [[1, 2, 3], [4, 5, 6]]
flat = [val for row in matrix for val in row]  # [1, 2, 3, 4, 5, 6]
Генераторные выражения (Generator Expressions)
Синтаксис в круглых скобках (...). В отличие от list comprehension, НЕ создает список в оперативной памяти!
🐍 python 
# Экономия гигабайт памяти на больших данных:
gen = (x**2 for x in range(100_000_000))  # занимает ~100 байт!
next(gen)   # 0
next(gen)   # 1

# Использование прямо внутри агрегирующих функций:
total_even_sum = sum(x for x in range(1000) if x % 2 == 0)
Если данные нужны только для однократного прохода или подсчета суммы/минимума — всегда используйте генераторные выражения вместо списков.

Условия и Pattern Matching

Основы языка
Условные конструкции if / elif / else
Ветвление логики и тернарный оператор.
🐍 python 
status = 200

if status == 200:
    msg = "OK"
elif 400 <= status < 500:  # Цепочка сравнений (Chained comparisons)
    msg = "Client Error"
else:
    msg = "Other"

# Тернарный оператор (однострочный if)
label = "доступ разрешен" if is_admin else "вход воспрещен"
Сопоставление с шаблоном: match / case (Python 3.10+) PYTHON 3.10+
Структурный Pattern Matching. Намного мощнее обычного switch/case из C++/Java.
🐍 python 
def handle_command(cmd):
    match cmd:
        case "quit" | "exit":                 # Или через |
            print("Выход из системы")
        case ["move", ("up" | "down" as direction), int(steps)]:
            print(f"Движение {direction} на {steps} шагов")
        case {"user": str(name), "role": "admin"}:  # Деструктуризация словаря
            print(f"Приветствуем администратора {name}")
        case int(x) if x > 0:                 # Guard-условие
            print(f"Положительное число: {x}")
        case _:                                # Дефолтный случай (wildcard)
            print("Команда не распознана")

Циклы (loops)

Декораторы

Продвинутый Python
Декораторы функций и @functools.wraps ПРОДВИНУТОЕ
Декоратор — это функция высшего порядка, принимающая функцию и возвращающая её модифицированную версию.
🐍 python 
from functools import wraps
import time

def timer(func):
    @wraps(func)  # ОБЯЗАТЕЛЬНО: сохраняет имя, docstring и метаданные оригинальной функции!
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - t0
        print(f"[{func.__name__}] выполнена за {elapsed:.4f} сек")
        return result
    return wrapper

@timer
def download_data(count):
    time.sleep(0.1)
    return [1] * count

download_data(100)
Декоратор с аргументами (Фабрика декораторов)
Три уровня вложенности: фабрика -> декоратор -> обертка.
🐍 python 
from functools import wraps

def repeat(num_times=3):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(num_times):
                res = func(*args, **kwargs)
            return res
        return wrapper
    return decorator

@repeat(num_times=2)
def ping():
    print("Понг!")
Порядок нескольких декораторов: @dec1 затем @dec2 выполняется как dec1(dec2(func)) — снизу вверх при создании, сверху вниз при вызове!

        self.username = username
        self._age = age      # _protected: соглашение об инкапсуляции

    def age(self):

    def from_string(cls, data_str: str):

        name, age = data_str.split(":")
        return cls(name, int(age))

Асинхронность и Параллелизм

Продвинутый Python
Асинхронное программирование: asyncio СОВРЕМЕННО
Однопоточная асинхронность для I/O-bound задач (сеть, базы данных, API).
🐍 python 
import asyncio

async def fetch_data(service_id: int, delay: float) -> dict:
    print(f"Запрос {service_id} начат...")
    await asyncio.sleep(delay)  # неблокирующее ожидание! (НЕ time.sleep!)
    return {"id": service_id, "status": 200}

async def main():
    # Вариант 1: asyncio.gather для параллельного запуска
    results = await asyncio.gather(
        fetch_data(1, 0.2),
        fetch_data(2, 0.1),
        fetch_data(3, 0.3)
    )
    print(f"Получено: {results}")

    # Вариант 2: современный TaskGroup (Python 3.11+) — гарантирует отмену при ошибках
    async with asyncio.TaskGroup() as tg:
        t1 = tg.create_task(fetch_data(10, 0.1))
        t2 = tg.create_task(fetch_data(20, 0.1))
    print(t1.result(), t2.result())

# Точка входа для запуска асинхронного цикла:
asyncio.run(main())

🐍 python 
# Установка: pip install "fastapi[standard]" uvicorn
from fastapi import FastAPI, HTTPException, status, Depends
from pydantic import BaseModel, Field

app = FastAPI(title="Store API", version="1.0.0")

# 1. Pydantic схема валидации тела запроса (Request Body)
class ItemCreate(BaseModel):
    name: str = Field(..., min_length=2, max_length=50)
    price: float = Field(..., gt=0, description="Цена должна быть > 0")
    is_offer: bool = False

class ItemResponse(ItemCreate):
    id: int

items_db: dict[int, dict] = {}

# 2. GET эндпоинт с Path и Query параметрами
@app.get("/items/{item_id}", response_model=ItemResponse)
async def get_item(item_id: int, detailed: bool = False):
    if item_id not in items_db:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Товар не найден")
    return items_db[item_id]

# 3. POST эндпоинт с автовалидацией Pydantic
@app.post("/items", response_model=ItemResponse, status_code=status.HTTP_201_CREATED)
async def create_item(item: ItemCreate):
    new_id = len(items_db) + 1
    saved = {"id": new_id, **item.model_dump()}
    items_db[new_id] = saved
    return saved

# Запуск в консоли: uvicorn main:app --reload
# Документация доступна в браузере по адресу: http://127.0.0.1:8000/docs
