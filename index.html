Короткая запись для создания коллекций из итерируемых объектов.

# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# Перевернуть словарь (значения → ключи)

# Только нужные ключи

# Generator expression — круглые скобки, не хранит всё в памяти

total = sum(x**2 for x in range(100))
Generator expression экономит память при работе с большими данными — вычисляет по одному элементу, а не хранит всё сразу.

Декоратор — функция, которая принимает другую функцию и добавляет к ней поведение.
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"Вызов {func.__name__}")
        result = func(*args, **kwargs)
        print(f"Готово, результат: {result}")
        return result
    return wrapper

@logger                   # синтаксический сахар для greet = logger(greet)
def greet(name):
    return f"Привет, {name}!"

Исключения (try / except)

Базовая структура
try:
    x = int(input("Число: "))
    result = 100 / x
except ValueError:
    print("Введи число, а не текст!")
except ZeroDivisionError:
    print("На ноль нельзя!")
except (TypeError, RuntimeError) as e:  # несколько + захват

    print(f"Результат: {result}")
finally:                      # выполняется ВСЕГДА
    print("Блок выполнен")

ООП — Классы

Базовый класс
class Person:
    species = "Homo sapiens"   # атрибут класса (общий для всех)

    def __init__(self, name: str, age: int):   # конструктор
        self.name = name    # атрибут экземпляра
        self.age  = age
        self._secret = "private"  # _ условно приватный

    def greet(self) -> str:       # метод экземпляра
        return f"Я {self.name}, мне {self.age}"

    def __repr__(self):             # repr для разработчика
        return f"Person({self.name!r}, {self.age})"

    def __str__(self):              # str для пользователя
        return self.name

    @classmethod
    def from_birth_year(cls, name, year):  # альтернативный конструктор
        return cls(name, 2024 - year)

    @staticmethod
    def validate_age(age):         # не имеет доступа к self/cls
        return 0 <= age <= 150

p = Person("Маша", 20)
p.greet()
Person.from_birth_year("Ваня", 2000)
Наследование
class Student(Person):       # наследует от Person
    def __init__(self, name, age, university):
        super().__init__(name, age)  # вызвать __init__ родителя
        self.university = university

    def greet(self):              # переопределить метод родителя
        base = super().greet()
        return f"{base}, учусь в {self.university}"

s = Student("Ваня", 19, "МГУ")
isinstance(s, Person)    # True  — Student это тоже Person
issubclass(Student, Person) # True

class Circle:

    @radius.setter
    def radius(self, value):    # с валидацией
        if value < 0: raise ValueError("Радиус не отрицательный")
        self._radius = value

    @property
    def area(self):             # вычисляемый атрибут
        return 3.14159 * self._radius ** 2

c = Circle(5)
c.radius     # 5
c.radius = 10
c.area       # 314.159

Генераторы и итераторы

Функция-генератор (yield)
Генератор вычисляет значения лениво — по одному при каждом вызове next(). Экономит память.
def countdown(n):
    while n > 0:
        yield n      # приостановить и вернуть n
        n -= 1

for x in countdown(5):
    print(x)   # 5, 4, 3, 2, 1

# Бесконечный генератор
def integers(start=0):
    while True:
        yield start
        start += 1

gen = integers()
next(gen)   # 0
next(gen)   # 1
